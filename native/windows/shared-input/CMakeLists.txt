cmake_minimum_required(VERSION 3.20)
project(FramesSharedInput LANGUAGES C CXX ASM_MASM)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Shared input requires Windows x64 MSVC")
endif()
set(EASY "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/frames-protoinput/lib/EasyHook")
file(GLOB LOCAL "${EASY}/DriverShared/LocalHook/*.c" "${EASY}/DriverShared/Rtl/*.c" "${EASY}/DriverShared/Disassembler/libudis86/*.c")
set(CORE ${LOCAL} "${EASY}/EasyHookDll/dllmain.c" "${EASY}/EasyHookDll/LocalHook/acl.c"
  "${EASY}/EasyHookDll/LocalHook/debug.cpp" "${EASY}/EasyHookDll/Rtl/memory.c")
# The pinned ProtoInput fork statically links EasyHook and compiles its C files as C++.
set_source_files_properties("${EASY}/EasyHookDll/dllmain.c" PROPERTIES LANGUAGE CXX)
add_library(frames_easyhook STATIC ${CORE} "${EASY}/DriverShared/ASM/HookSpecific_x64.asm")
target_include_directories(frames_easyhook PRIVATE "${EASY}/EasyHookDll" "${EASY}/Public" "${EASY}/DriverShared")
target_compile_definitions(frames_easyhook PRIVATE _ALLOW_RTCc_IN_STL _CRT_SECURE_NO_WARNINGS)
target_compile_options(frames_easyhook PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W0>")
add_library(FramesInput SHARED adapter.cpp broker.cpp process_audio.cpp)
target_include_directories(FramesInput PRIVATE "${EASY}/Public")
target_compile_features(FramesInput PRIVATE cxx_std_17)
target_compile_definitions(FramesInput PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(FramesInput PRIVATE frames_easyhook user32 gdi32 advapi32 bcrypt psapi ole32 mmdevapi)
set_property(TARGET FramesInput frames_easyhook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_executable(shared_input_fixture fixture.cpp)
target_compile_features(shared_input_fixture PRIVATE cxx_std_17)
target_compile_definitions(shared_input_fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(shared_input_fixture PRIVATE user32 gdi32)
add_executable(shared_audio_fixture audio_fixture.cpp)
target_compile_features(shared_audio_fixture PRIVATE cxx_std_17)
target_compile_definitions(shared_audio_fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(shared_audio_fixture PRIVATE winmm)
add_executable(shared_ipc_fixture ipc_fixture.cpp)
target_compile_features(shared_ipc_fixture PRIVATE cxx_std_17)
target_compile_definitions(shared_ipc_fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
